# Live Score Game-By-Game Detail Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Let a user click a live score card and see the match's game-by-game history (who served, final point score, set by set), persisted permanently in the DB, for both in-progress and finished matches.

**Architecture:** Every poll tick, diff the newly-parsed `LiveMatchState` against the previous tick's state per match; when a game closes, write one row to a new `match_game_events` table. A new `GET /api/scores/{match_id}/games` endpoint returns that history plus match metadata (from the live cache or `FinishedMatch`). The frontend opens a modal on card click, fetching that endpoint.

**Tech Stack:** FastAPI, SQLAlchemy (async) + Alembic, Pydantic, Vue 3 `<script setup>` + Pinia (existing store untouched), Vitest not used (no frontend test infra exists in this repo).

**Spec:** `docs/superpowers/specs/2026-09-08-live-score-game-log-design.md`

## Global Constraints

- Singles matches only — same universe as `StatsService`/`FinishedMatch` (real player names via `_is_real_player_name`, both Elo present, singles `player_config`).
- Game-by-game granularity, not point-by-point — see spec's "Source constraint" section.
- DB writes for the new table use `session.add()` + `flush()`/`commit()` inside `try/except IntegrityError`, never dialect-specific `ON CONFLICT`/`INSERT OR IGNORE` — matches `stats_service._try_finish_match`.
- `GameServer.live_state` / the parsed `LiveMatchState` for a given tick must be computed once per server and shared between the win-probability step and the new game-log step — never parsed twice in the same tick.
- No new frontend test framework — this repo has zero Vue component tests today (bare `vitest` script, no config, no `@vue/test-utils`). Verify the frontend change manually in the browser instead.
- No foreign key between `match_game_events` and `finished_matches` — joined only by the `match_id` string at query time.

```python
"""Match game event database model.

Stores the game-by-game history of a live or finished match, written
incrementally as each game completes (see the tick-diff service in
game_log_service.py). Independent of FinishedMatch — joined only by
match_id at query time, so a match keeps whatever game history it
accrued even if it's abandoned before qualifying as "finished".
"""

from datetime import datetime

from sqlalchemy import Boolean, DateTime, Integer, String, UniqueConstraint, func
from sqlalchemy.orm import Mapped, mapped_column

from app.core.database import Base


class MatchGameEvent(Base):
    """One completed game within a live or finished match."""

    __tablename__ = "match_game_events"

    id: Mapped[int] = mapped_column(primary_key=True, autoincrement=True)
    match_id: Mapped[str] = mapped_column(String, index=True)
    set_number: Mapped[int] = mapped_column(Integer)
    game_number: Mapped[int] = mapped_column(Integer)
    p1_games: Mapped[int] = mapped_column(Integer)
    p2_games: Mapped[int] = mapped_column(Integer)
    winner: Mapped[int] = mapped_column(Integer)
    server: Mapped[int | None] = mapped_column(Integer, nullable=True)
    final_point_score: Mapped[str | None] = mapped_column(String, nullable=True)
    is_tiebreak: Mapped[bool] = mapped_column(Boolean, default=False)

    created_at: Mapped[datetime] = mapped_column(
        DateTime, server_default=func.now(), index=True
    )

    __table_args__ = (
        UniqueConstraint("match_id", "set_number", "game_number", name="uq_match_game_event"),
    )
```

```python
"""Game-by-game live score history.

Diffs consecutive polling snapshots (LiveMatchState) to detect completed
games and persists them to `match_game_events`, incrementally, for both
live and finished matches. Game-by-game, not point-by-point — the TE4
master server only exposes a current-state snapshot per poll, no event
log (see docs/superpowers/specs/2026-09-08-live-score-game-log-design.md).
"""

from dataclasses import dataclass

from sqlalchemy.exc import IntegrityError

from app.core.database import get_session_factory
from app.core.logging import get_logger
from app.models.match_game_event import MatchGameEvent
from app.services.score_parser import LiveMatchState

logger = get_logger("game_log_service")


@dataclass(frozen=True)
class GameEventFields:
    set_number: int
    game_number: int
    p1_games: int
    p2_games: int
    winner: int
    server: int | None
    final_point_score: str | None
    is_tiebreak: bool


def diff_game(previous: LiveMatchState, current: LiveMatchState) -> GameEventFields | None:
    """Returns the game that closed between `previous` and `current`'s
    snapshots, or None if no game completed this tick.

    `previous.current_set_games` is always the pre-game game count in both
    branches below (the "set closed" branch never bumped it — that
    snapshot's set was still in progress — and the "game closed mid-set"
    branch diffs against it directly), so the winner's +1 is applied
    exactly once, after the branch, not per-branch.
    """
    prev_sets_total = sum(previous.sets_won)
    curr_sets_total = sum(current.sets_won)

    if curr_sets_total > prev_sets_total:
        # The game in progress last tick closed out the set (straight
        # games or tiebreak) — current_set_games may have already reset
        # to 0/0 for the next set by this tick.
        set_number = prev_sets_total + 1
        winner = 1 if current.sets_won[0] > previous.sets_won[0] else 2
    elif (
        sum(current.current_set_games) > sum(previous.current_set_games)
        and curr_sets_total == prev_sets_total
    ):
        set_number = curr_sets_total + 1
        winner = 1 if current.current_set_games[0] > previous.current_set_games[0] else 2
    else:
        return None

    p1, p2 = previous.current_set_games
    if winner == 1:
        p1 += 1
    else:
        p2 += 1

    final_point_score = (
        f"{previous.current_points[0]}-{previous.current_points[1]}"
        if previous.current_points
        else None
    )

    return GameEventFields(
        set_number=set_number,
        game_number=p1 + p2,
        p1_games=p1,
        p2_games=p2,
        winner=winner,
        server=previous.server,
        final_point_score=final_point_score,
        is_tiebreak=previous.is_tiebreak,
    )
```

```python
class GameLogService:
    """Per-tick tracker of live match state, writing completed games to the DB."""

    def __init__(self) -> None:
        self._previous: dict[str, LiveMatchState] = {}

    async def record_tick(self, current_states: dict[str, LiveMatchState]) -> None:
        """Diff each match's state against last tick's and write any game
        that closed. Called once per poll tick from ScraperService.fetch_servers."""
        for match_id, current in current_states.items():
            previous = self._previous.get(match_id)
            if previous is not None:
                event = diff_game(previous, current)
                if event is not None:
                    await self._insert_event(match_id, event)
            self._previous[match_id] = current

        # Drop state for matches no longer present this tick — this
        # service only needs to stop tracking them, not run
        # stats_service's rename/finish detection, so a plain key-set
        # difference against this tick's own dict is enough.
        for stale_id in self._previous.keys() - current_states.keys():
            del self._previous[stale_id]

    async def _insert_event(self, match_id: str, event: GameEventFields) -> None:
        session_factory = get_session_factory()
        async with session_factory() as session:
            try:
                session.add(
                    MatchGameEvent(
                        match_id=match_id,
                        set_number=event.set_number,
                        game_number=event.game_number,
                        p1_games=event.p1_games,
                        p2_games=event.p2_games,
                        winner=event.winner,
                        server=event.server,
                        final_point_score=event.final_point_score,
                        is_tiebreak=event.is_tiebreak,
                    )
                )
                await session.commit()
            except IntegrityError:
                await session.rollback()
                logger.debug(
                    f"Duplicate game event ignored for {match_id} "
                    f"set {event.set_number} game {event.game_number}"
                )
            except Exception as e:
                await session.rollback()
                logger.error(f"Failed to record game event for {match_id}: {e}")


_game_log_service: GameLogService | None = None


def get_game_log_service() -> GameLogService:
    """Get the game-log service singleton."""
    global _game_log_service
    if _game_log_service is None:
        _game_log_service = GameLogService()
    return _game_log_service
```

```python
            p0, g, s_rate = rates
            server.win_probability = live_win_probability(
                g, s_rate, live_state, stats_service.sets_to_win(server)
            )

        # Concurrent so a burst of simultaneously-new matches costs one
        # round-trip's latency, not N sequential ones. return_exceptions=True
        # so one match's unexpected failure can never propagate and affect
        # any other match (or the whole tick's broadcast).
        win_prob_results = await asyncio.gather(
            *(_apply_win_probability(s) for s in singles_servers),
            return_exceptions=True,
        )
        for server, win_prob_result in zip(singles_servers, win_prob_results, strict=True):
            if isinstance(win_prob_result, BaseException):
                logger.error(
                    f"win-probability computation failed for {server.match_name}: {win_prob_result}"
                )

        # Track finished matches for stats, and record any game that
        # closed this tick — same gate as finished-match tracking, since
        # both are internal bookkeeping side effects, not part of the
        # public score payload itself.
        if track_stats and servers:
            finished = await stats_service.track_matches(servers)
            if finished > 0:
                logger.info(f"Detected {finished} finished matches")

            await get_game_log_service().record_tick(singles_live_states)
```

```typescript
/**
 * Composable for fetching a match's game-by-game history (who served,
 * final point score of each game, set by set). Public endpoint, no auth.
 */
import { ref } from 'vue'
import { apiUrl } from '@/config/api'

export interface GameEntry {
  set_number: number
  game_number: number
  p1_games: number
  p2_games: number
  winner: 1 | 2
  server: 1 | 2 | null
  final_point_score: string | null
  is_tiebreak: boolean
}

export interface MatchGameLog {
  match_id: string
  player1: string
  player2: string
  p1_elo: number | null
  p2_elo: number | null
  surface: string | null
  mod: string | null
  score: string | null
  is_live: boolean
  games: GameEntry[]
}

export function useMatchGameLog() {
  const log = ref<MatchGameLog | null>(null)
  const isLoading = ref(false)
  const error = ref<string | null>(null)

  let requestId = 0

  async function fetchGameLog(matchId: string): Promise<void> {
    const thisRequest = ++requestId
    isLoading.value = true
    error.value = null

    try {
      const response = await fetch(apiUrl(`/api/scores/${encodeURIComponent(matchId)}/games`))
      if (thisRequest !== requestId) return

      if (response.status === 404) {
        throw new Error('Match not found.')
      }
      if (!response.ok) {
        throw new Error(`HTTP error ${response.status}`)
      }

      log.value = await response.json()
    } catch (e: any) {
      if (thisRequest !== requestId) return
      error.value = e.message
      log.value = null
      console.error('Failed to fetch match game log:', e)
    } finally {
      if (thisRequest === requestId) {
        isLoading.value = false
      }
    }
  }

  function clearGameLog(): void {
    requestId++
    log.value = null
    error.value = null
    isLoading.value = false
  }

  return { log, isLoading, error, fetchGameLog, clearGameLog }
}
```

```typescript
let pollHandle: ReturnType<typeof setInterval> | null = null

function stopPolling() {
  if (pollHandle !== null) {
    clearInterval(pollHandle)
    pollHandle = null
  }
}

watch(
  () => [props.open, props.matchId] as const,
  ([open, matchId]) => {
    stopPolling()
    if (!open || !matchId) {
      clearGameLog()
      return
    }
    fetchGameLog(matchId)
  }
)

// Once the fetch resolves and the match turns out to be live, start
// polling for new games. Re-evaluated on every `log` change so polling
// stops on its own once the match finishes (is_live flips to false on
// the next fetch after the match leaves the live cache).
watch(log, (current) => {
  stopPolling()
  if (current?.is_live && props.open) {
    pollHandle = setInterval(() => {
      if (props.matchId) fetchGameLog(props.matchId)
    }, POLL_INTERVAL_MS)
  }
})

onUnmounted(stopPolling)
```
